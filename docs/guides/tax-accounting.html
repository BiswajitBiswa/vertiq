Working example

from datetime import datetime, timedelta
from vertiq import plugins, rules
from vertiq.mapping import Mapping, apply, repair
from vertiq.tasks import predict_event, predict_value

# rows: your export as a list of dicts, e.g. list(csv.DictReader(open("export.csv")))
mapping = Mapping.parse({"entity": "GSTIN", "time": "Abgabedatum", "type": "Return Type", "value": "Tax Amount", "group": "Jurisdiction", "attrs": ["Status", "Filed By", "Due Date", "Filed On"], "date_format": "%Y%m%d"})
m, notes = repair(mapping, rows)      # the rows settle column names, value types and the date format
ds, problems = apply(m, rows)         # any row that didn't load is named, never guessed

@plugins.feature("tax")
def late_filings(entity, before, cut):   # only events before the cutoff: no peeking
    return {"late_before": sum(1 for e in before if e.attrs.get("Status") == "late")}

task = predict_event(ds, target="GSTR-3B", window_days=30, pack="tax")
print(task.status, task.report())     # "trained" only if it beat the baseline on held-out entities
if task.status == "trained":
    print(task.score("29ABCDE1234F1Z5"))  # the chance, and what raises or lowers it

flags = rules.run(ds, [
    rules.overdue("Due Date", "Filed On", tolerance_days=0),   # returns filed after the due date
])
print(flags["summary"])

This exact code runs in the test suite (tests/test_guides.py) against generated rows in these column names.

Domain knowledge: late_filings

Clients who filed late before tend to file late again.

Checks

  • rules.overdue("Due Date", "Filed On", tolerance_days=0): returns filed after the due date

Before you rely on it

  • Dates in due and done columns must be ISO dates (2026-03-31) for overdue().
